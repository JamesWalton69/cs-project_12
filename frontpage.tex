% Place text at the original PDF baseline. The third argument is a horizontal
% scale factor; vertical letter proportions are left untouched.
% Usage: \PlaceText{x}{baseline-y}{x-scale}{unused-height}{font commands}{text}
\newcommand{\PlaceText}[6]{%
  \node[anchor=base west,inner sep=0pt,outer sep=0pt]
    at ([xshift=#1pt,yshift=-#2pt]current page.north west)
    {\scalebox{#3}[1]{{#5 #6}}};%
}
